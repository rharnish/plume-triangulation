#     uv venv --python 3.12 .venv-train
#     uv pip install --python .venv-train/bin/python -r requirements-train.txt \
#         --extra-index-url https://download.pytorch.org/whl/cu126 --index-strategy unsafe-best-match
#
# torch is the same cu126 pin as requirements-masks.txt, for the same reason (GTX 1070,
# sm_61, carried by the sm_60 image -- see that file). ultralytics matches
# requirements-edge.txt so weights trained here export to Core ML the same way.
